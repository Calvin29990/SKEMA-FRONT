import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

stock_data["Daily_Return"] = stock_data["Close"].pct_change()

stock_data["Cumulative_Return"] = (
    stock_data["Close"] /
    stock_data["Close"].iloc[0] - 1
) * 100

stock_data["MA20"] = stock_data["Close"].rolling(20).mean()

volatility = stock_data["Daily_Return"].std() * np.sqrt(252) * 100

sharpe_ratio = (
    stock_data["Daily_Return"].mean() * 252
) / (
    stock_data["Daily_Return"].std() * np.sqrt(252)
)

running_max = stock_data["Close"].cummax()

stock_data["Drawdown"] = (
    stock_data["Close"] - running_max
) / running_max * 100

max_drawdown = stock_data["Drawdown"].min()

total_return = stock_data["Cumulative_Return"].iloc[-1]

print("Return (%) =", round(total_return,2))
print("Volatility (%) =", round(volatility,2))
print("Sharpe Ratio =", round(sharpe_ratio,2))

# RISK RETURN MATRIX

benchmarks = [
    "NVDA",
    "AAPL",
    "MSFT",
    "GOOGL",
    "AMZN",
    "META",
    "TSLA"
]

risks = []
returns = []

for ticker in benchmarks:

    temp = yf.download(
        ticker,
        start=start_date,
        end=end_date,
        multi_level_index=False
    )

    temp["Return"] = temp["Close"].pct_change()

    risk = temp["Return"].std() * np.sqrt(252) * 100

    ret = (
        temp["Close"].iloc[-1] /
        temp["Close"].iloc[0] - 1
    ) * 100

    risks.append(risk)
    returns.append(ret)

plt.figure(figsize=(10,7))

for i, ticker in enumerate(benchmarks):

    plt.scatter(
        risks[i],
        returns[i],
        s=250
    )

    plt.annotate(
        ticker,
        (risks[i], returns[i])
    )

plt.title("Risk-Return Matrix")
plt.xlabel("Risk (Volatility %)")
plt.ylabel("Return (%)")
plt.grid(True)

plt.show()
